```java
package com.example.ninebox.model;

public class Employee {
    private String name;
    private int potential;    // 1..10
    private int performance;  // 1..10

    public Employee(String name, int potential, int performance) {
        this.name = name;
        this.potential = potential;
        this.performance = performance;
    }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }

    public int getPotential() { return potential; }
    public void setPotential(int potential) { this.potential = potential; }

    public int getPerformance() { return performance; }
    public void setPerformance(int performance) { this.performance = performance; }

    /** 0=низкий(1-3), 1=средний(4-7), 2=высокий(8-10) */
    public int getPotentialGroup() { return toGroup(potential); }
    public int getPerformanceGroup() { return toGroup(performance); }

    private static int toGroup(int score) {
        if (score <= 3) return 0;
        if (score <= 7) return 1;
        return 2;
    }

    /** Типичный балл для группы: 2, 5, 9 — используется при drop */
    public static int groupToScore(int group) {
        return switch (group) {
            case 0 -> 2;
            case 1 -> 5;
            default -> 9;
        };
    }

    /** Критерий квадрата, в котором находится сотрудник */
    public NineBoxCriterion getCriterion() {
        return NineBoxCatalog.get(getPotentialGroup(), getPerformanceGroup());
    }

    /** Название квадрата (для таблицы) */
    public String getBoxName() {
        NineBoxCriterion c = getCriterion();
        return c == null ? "" : c.getName();
    }
}
```

```java
package com.example.ninebox.storage;

import com.example.ninebox.model.Employee;
import com.example.ninebox.model.NineBoxCatalog;
import com.example.ninebox.model.NineBoxCriterion;
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

import java.io.*;
import java.util.ArrayList;
import java.util.List;

public class XlsxStorage {
    private static final String FILE = "employees.xlsx";

    public static List<Employee> load() {
        List<Employee> list = new ArrayList<>();
        File f = new File(FILE);
        if (!f.exists()) return list;

        try (Workbook wb = new XSSFWorkbook(new FileInputStream(f))) {
            Sheet sheet = wb.getSheet("Сотрудники");
            if (sheet == null) sheet = wb.getSheetAt(0);
            for (int i = 1; i <= sheet.getLastRowNum(); i++) {
                Row r = sheet.getRow(i);
                if (r == null) continue;
                String name = getString(r.getCell(0));
                int pot = (int) getNum(r.getCell(1));
                int perf = (int) getNum(r.getCell(2));
                if (!name.isBlank()) list.add(new Employee(name, pot, perf));
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
        return list;
    }

    public static void saveAndExport(List<Employee> list, String file) {
        try (Workbook wb = new XSSFWorkbook()) {
            writeEmployeeSheet(wb, list);
            writeMatrixSheet(wb, list);
            writeToFile(wb, file);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    public static void exportMatrixToExisting(List<Employee> list, String file) {
        File f = new File(file);
        try {
            Workbook wb = f.exists()
                    ? new XSSFWorkbook(new FileInputStream(f))
                    : new XSSFWorkbook();
            Sheet old = wb.getSheet("Матрица 9 квадратов");
            if (old != null) wb.removeSheetAt(wb.getSheetIndex(old));
            writeMatrixSheet(wb, list);
            writeToFile(wb, file);
            wb.close();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static void writeEmployeeSheet(Workbook wb, List<Employee> list) {
        Sheet sheet = wb.createSheet("Сотрудники");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("Имя");
        header.createCell(1).setCellValue("Потенциал");
        header.createCell(2).setCellValue("Результативность");
        header.createCell(3).setCellValue("Квадрат");

        int rowIdx = 1;
        for (Employee e : list) {
            Row r = sheet.createRow(rowIdx++);
            r.createCell(0).setCellValue(e.getName());
            r.createCell(1).setCellValue(e.getPotential());
            r.createCell(2).setCellValue(e.getPerformance());
            r.createCell(3).setCellValue(e.getBoxName());
        }
        for (int i = 0; i < 4; i++) sheet.setColumnWidth(i, 5000);
    }

    private static void writeMatrixSheet(Workbook wb, List<Employee> list) {
        Sheet matrix = wb.createSheet("Матрица 9 квадратов");

        String[] perfLabels = {"Низкая рез.", "Средняя рез.", "Высокая рез."};
        String[] potLabels  = {"Низкий потенциал", "Средний потенциал", "Высокий потенциал"};

        CellStyle headerStyle = wb.createCellStyle();
        Font bold = wb.createFont();
        bold.setBold(true);
        headerStyle.setFont(bold);

        Row top = matrix.createRow(0);
        Cell corner = top.createCell(0);
        corner.setCellValue("Потенциал \\ Результат");
        corner.setCellStyle(headerStyle);
        for (int c = 0; c < 3; c++) {
            Cell cell = top.createCell(c + 1);
            cell.setCellValue(perfLabels[c]);
            cell.setCellStyle(headerStyle);
        }

        for (int p = 0; p < 3; p++) {
            int pg = 2 - p;
            Row row = matrix.createRow(p + 1);
            Cell left = row.createCell(0);
            left.setCellValue(potLabels[pg]);
            left.setCellStyle(headerStyle);

            for (int rg = 0; rg < 3; rg++) {
                StringBuilder sb = new StringBuilder();
                for (Employee e : list) {
                    if (e.getPotentialGroup() == pg && e.getPerformanceGroup() == rg) {
                        if (sb.length() > 0) sb.append("\n");
                        sb.append(e.getName())
                          .append(" (").append(e.getPotential())
                          .append("/").append(e.getPerformance()).append(")");
                    }
                }
                Cell cell = row.createCell(rg + 1);
                cell.setCellValue(sb.toString());

                CellStyle style = wb.createCellStyle();
                style.setFillPattern(FillPatternType.SOLID_FOREGROUND);
                style.setFillForegroundColor(switch (colorOf(pg, rg)) {
                    case "green"  -> IndexedColors.LIGHT_GREEN.getIndex();
                    case "yellow" -> IndexedColors.LIGHT_YELLOW.getIndex();
                    default       -> IndexedColors.ROSE.getIndex();
                });
                style.setWrapText(true);
                cell.setCellStyle(style);
            }
        }
        for (int c = 0; c < 4; c++) matrix.setColumnWidth(c, 7000);

        // Блок критериев
        int startRow = 6;
        Row title = matrix.createRow(startRow++);
        Cell titleCell = title.createCell(0);
        titleCell.setCellValue("Критерии оценки 9 квадратов");
        titleCell.setCellStyle(headerStyle);

        String[] headers = {"Квадрат", "Кто это", "Риски", "Что делать"};
        Row hdr = matrix.createRow(startRow++);
        for (int i = 0; i < headers.length; i++) {
            Cell c = hdr.createCell(i);
            c.setCellValue(headers[i]);
            c.setCellStyle(headerStyle);
        }

        for (int pg = 2; pg >= 0; pg--) {
            for (int rg = 0; rg <= 2; rg++) {
                NineBoxCriterion c = NineBoxCatalog.get(pg, rg);
                if (c == null) continue;
                Row r = matrix.createRow(startRow++);
                r.createCell(0).setCellValue(c.getName());
                r.createCell(1).setCellValue(c.getWho());
                r.createCell(2).setCellValue(c.getRisks());
                r.createCell(3).setCellValue(c.getActions());
                for (int i = 0; i < 4; i++) {
                    Cell cell = r.getCell(i);
                    if (cell == null) cell = r.createCell(i);
                    CellStyle wrap = wb.createCellStyle();
                    wrap.setWrapText(true);
                    cell.setCellStyle(wrap);
                }
            }
        }
        matrix.setColumnWidth(1, 14000);
        matrix.setColumnWidth(2, 14000);
        matrix.setColumnWidth(3, 14000);
    }

    private static String colorOf(int pg, int rg) {
        if (pg == 0) return rg == 0 ? "red" : "yellow";
        if (pg == 1) return rg == 0 ? "red" : (rg == 1 ? "yellow" : "green");
        return rg == 0 ? "yellow" : "green";
    }

    private static void writeToFile(Workbook wb, String file) throws IOException {
        try (FileOutputStream out = new FileOutputStream(file)) {
            wb.write(out);
        }
    }

    private static String getString(Cell c) {
        return c == null ? "" : switch (c.getCellType()) {
            case STRING -> c.getStringCellValue();
            case NUMERIC -> String.valueOf((long) c.getNumericCellValue());
            default -> "";
        };
    }

    private static double getNum(Cell c) {
        return c == null ? 0 : c.getNumericCellValue();
    }
}
```

```java
package com.example.ninebox.controller;

import com.example.ninebox.model.Employee;
import com.example.ninebox.model.NineBoxCatalog;
import com.example.ninebox.model.NineBoxCriterion;
import com.example.ninebox.storage.XlsxStorage;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.geometry.Insets;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.input.*;
import javafx.scene.layout.*;

import java.util.ArrayList;

public class MainController {

    @FXML private TextField nameField, potentialField, performanceField;
    @FXML private GridPane matrixGrid;
    @FXML private TableView<Employee> employeeTable;
    @FXML private TableColumn<Employee, String> nameCol;
    @FXML private TableColumn<Employee, Integer> potCol, perfCol;
    @FXML private TableColumn<Employee, String> boxCol;
    @FXML private Label infoTitle, infoWho, infoRisks, infoActions;

    private final ObservableList<Employee> employees = FXCollections.observableArrayList();
    private final VBox[][] cells = new VBox[3][3];

    private static final String[] POT_LABELS  = {"Низкий потенциал", "Средний потенциал", "Высокий потенциал"};
    private static final String[] PERF_LABELS = {"Низкая рез.", "Средняя рез.", "Высокая рез."};
    private static final DataFormat EMPLOYEE_FORMAT = new DataFormat("application/x-employee");

    private static final String[][] CELL_COLORS = {
            {"#f8d7da", "#fff3cd", "#fff3cd"},
            {"#f8d7da", "#fff3cd", "#d4edda"},
            {"#fff3cd", "#d4edda", "#d4edda"}
    };

    @FXML
    public void initialize() {
        nameCol.setCellValueFactory(new PropertyValueFactory<>("name"));
        potCol.setCellValueFactory(new PropertyValueFactory<>("potential"));
        perfCol.setCellValueFactory(new PropertyValueFactory<>("performance"));
        boxCol.setCellValueFactory(new PropertyValueFactory<>("boxName"));

        buildMatrix();
        employees.addAll(XlsxStorage.load());
        refreshMatrix();
        employeeTable.setItems(employees);

        employeeTable.getSelectionModel().selectedItemProperty().addListener((obs, o, n) -> {
            if (n != null) showCriterion(n.getPotentialGroup(), n.getPerformanceGroup());
        });
    }

    private void buildMatrix() {
        for (int p = 0; p < 3; p++) {
            for (int r = 0; r < 3; r++) {
                final int pg = 2 - p;
                final int rg = r;

                VBox cell = new VBox(4);
                cell.setPadding(new Insets(6));
                cell.setStyle("-fx-border-color: #888; -fx-border-radius: 4; "
                        + "-fx-background-color: " + CELL_COLORS[pg][rg] + ";");
                cell.setMinSize(160, 140);

                NineBoxCriterion crit = NineBoxCatalog.get(pg, rg);
                Label header = new Label(crit.getName());
                header.setStyle("-fx-font-weight: bold; -fx-font-size: 11px;");
                cell.getChildren().add(header);

                cell.setOnDragOver(ev -> {
                    if (ev.getDragboard().hasContent(EMPLOYEE_FORMAT))
                        ev.acceptTransferModes(TransferMode.MOVE);
                    ev.consume();
                });

                cell.setOnDragDropped(ev -> {
                    Dragboard db = ev.getDragboard();
                    if (db.hasContent(EMPLOYEE_FORMAT)) {
                        Employee e = (Employee) db.getContent(EMPLOYEE_FORMAT);
                        e.setPotential(Employee.groupToScore(pg));
                        e.setPerformance(Employee.groupToScore(rg));
                        refreshMatrix();
                        employeeTable.refresh();
                        showCriterion(pg, rg);
                        ev.setDropCompleted(true);
                    }
                    ev.consume();
                });

                cell.setOnMouseClicked(ev -> showCriterion(pg, rg));

                cells[pg][rg] = cell;
                matrixGrid.add(cell, r, p);
            }
        }
    }

    private void refreshMatrix() {
        for (int p = 0; p < 3; p++)
            for (int r = 0; r < 3; r++)
                cells[p][r].getChildren().removeIf(n -> n.getUserData() != null);

        for (Employee e : employees) {
            VBox cell = cells[e.getPotentialGroup()][e.getPerformanceGroup()];
            Label card = new Label(e.getName() + " (" + e.getPotential() + "/" + e.getPerformance() + ")");
            card.setUserData(e);
            card.setStyle("-fx-background-color: #ffffffcc; -fx-padding: 4; "
                    + "-fx-background-radius: 4; -fx-cursor: hand; -fx-border-color: #999;");
            card.setMaxWidth(Double.MAX_VALUE);

            card.setOnDragDetected(ev -> {
                Dragboard db = card.startDragAndDrop(TransferMode.MOVE);
                ClipboardContent cc = new ClipboardContent();
                cc.put(EMPLOYEE_FORMAT, e);
                db.setContent(cc);
                ev.consume();
            });

            cell.getChildren().add(card);
        }
    }

    private void showCriterion(int pg, int rg) {
        NineBoxCriterion c = NineBoxCatalog.get(pg, rg);
        if (c == null) return;
        infoTitle.setText(c.getName());
        infoWho.setText("Кто это: " + c.getWho());
        infoRisks.setText("Риски: " + c.getRisks());
        infoActions.setText("Что делать: " + c.getActions());
    }

    @FXML
    private void onAdd() {
        String name = nameField.getText().trim();
        int pot, perf;
        try {
            pot = Integer.parseInt(potentialField.getText().trim());
            perf = Integer.parseInt(performanceField.getText().trim());
        } catch (NumberFormatException ex) {
            showError("Потенциал и результативность должны быть числами 1–10");
            return;
        }
        if (name.isEmpty() || pot < 1 || pot > 10 || perf < 1 || perf > 10) {
            showError("Проверьте имя и баллы (1–10)");
            return;
        }
        employees.add(new Employee(name, pot, perf));
        nameField.clear(); potentialField.clear(); performanceField.clear();
        refreshMatrix();
    }

    @FXML
    private void onDelete() {
        Employee sel = employeeTable.getSelectionModel().getSelectedItem();
        if (sel != null) {
            employees.remove(sel);
            refreshMatrix();
        }
    }

    @FXML
    private void onSave() {
        XlsxStorage.saveAndExport(new ArrayList<>(employees), "employees.xlsx");
        new Alert(Alert.AlertType.INFORMATION, "Сохранено в employees.xlsx").showAndWait();
    }

    @FXML
    private void onExportMatrix() {
        XlsxStorage.exportMatrixToExisting(new ArrayList<>(employees), "employees.xlsx");
        new Alert(Alert.AlertType.INFORMATION, "Матрица экспортирована в employees.xlsx").showAndWait();
    }

    public void saveOnExit() {
        XlsxStorage.saveAndExport(new ArrayList<>(employees), "employees.xlsx");
    }

    private void showError(String msg) {
        new Alert(Alert.AlertType.WARNING, msg).showAndWait();
    }
}
```

IMAP_SERVER = "imap.yandex.ru"

USERNAME = "your_email@company.ru"
PASSWORD = "your_password"

mail = imaplib.IMAP4_SSL(IMAP_SERVER, IMAP_PORT)
mail.login(USERNAME, PASSWORD)
mail.select("inbox")
print("✅ Подключено")

# --- ПОИСК ВСЕХ ПИСЕМ ---
status, messages = mail.search(None, "ALL")
email_ids = messages[0].split()
print(f"📬 Найдено писем: {len(email_ids)}")

# --- СБОР ТЕМ ---
subjects = []
for i, e_id in enumerate(email_ids, start=1):
    # Забираем только заголовки, тело не нужно — это в разы быстрее
    status, msg_data = mail.fetch(e_id, "(BODY.PEEK[HEADER.FIELDS (SUBJECT)])")
    if status != "OK":
        subjects.append(f"{i}. <не удалось получить>")
        continue

    raw = msg_data[0][1]
    msg = email.message_from_bytes(raw)
    raw_subject = msg.get("Subject")

    if raw_subject is None:
        subject = "<без темы>"
    else:
        # Декодируем (может быть несколько частей в разных кодировках)
        parts = decode_header(raw_subject)
        subject = ""
        for part, encoding in parts:
            if isinstance(part, bytes):
                try:
                    subject += part.decode(encoding or "utf-8", errors="ignore")
                except (LookupError, TypeError):
                    subject += part.decode("utf-8", errors="ignore")
            else:
                subject += part

    subjects.append(f"{i}. {subject}")
    print(f"[{i}/{len(email_ids)}] {subject}")

# --- ЗАПИСЬ В ФАЙЛ ---
with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
    f.write("\n".join(subjects))

print(f"🏁 Готово! Темы сохранены в {OUTPUT_FILE}")










Чтобы выгрузить входящие письма с корпоративной почты в текстовый файл с помощью Python, нужно подключиться к почтовому серверу по протоколу IMAP. Это стандартный способ, который поддерживают большинство корпоративных почтовых систем (Яндекс 360, Mail.ru для бизнеса, Exchange, Gmail и др.).

🔑 Что понадобится

Прежде всего, вам нужны данные для подключения, которые обычно выдаёт IT-отдел:

· Адрес IMAP-сервера (например, imap.yandex.ru или outlook.office365.com).
· Порт (обычно 993 для защищённого соединения SSL).
· Логин и пароль от корпоративной почты.

Важно: Для некоторых сервисов (например, Gmail или Microsoft 365) может потребоваться создание пароля приложения в настройках безопасности аккаунта, так как обычный пароль может не подойти из-за двухфакторной аутентификации.

📝 Пример скрипта на Python

Вот готовый скрипт, который подключается к почте, перебирает все письма в папке «Входящие» и сохраняет каждое из них в отдельный .txt файл:

```python
import imaplib
import email
from email.header import decode_header
import os
import re

# --- НАСТРОЙКИ ПОДКЛЮЧЕНИЯ ---
IMAP_SERVER = "imap.yandex.ru"      # Адрес вашего IMAP-сервера
IMAP_PORT = 993                     # Порт (обычно 993 для SSL)
USERNAME = "your_email@company.ru"  # Ваш логин
PASSWORD = "your_password"          # Пароль или пароль приложения

# Папка для сохранения писем
OUTPUT_DIR = "emails_output"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# --- ПОДКЛЮЧЕНИЕ К СЕРВЕРУ ---
try:
    # Используем IMAP4_SSL для защищённого соединения
    mail = imaplib.IMAP4_SSL(IMAP_SERVER, IMAP_PORT)
    mail.login(USERNAME, PASSWORD)
    print("✅ Успешное подключение к почте.")
except Exception as e:
    print(f"❌ Ошибка подключения: {e}")
    exit(1)

# --- ВЫБОР ПАПКИ "ВХОДЯЩИЕ" ---
mail.select("inbox")

# --- ПОИСК ВСЕХ ПИСЕМ ---
# Критерий 'ALL' означает все письма. Можно заменить на 'UNSEEN' для непрочитанных.
status, messages = mail.search(None, "ALL")
if status != "OK":
    print("❌ Не удалось получить список писем.")
    mail.logout()
    exit(1)

# Получаем список ID писем (это байтовые строки, разделённые пробелами)
email_ids = messages[0].split()
print(f"📬 Найдено писем: {len(email_ids)}")

# --- ЦИКЛ ПО ПИСЬМАМ И СОХРАНЕНИЕ ---
for i, e_id in enumerate(email_ids, start=1):
    # Извлекаем тело письма в формате RFC822
    status, msg_data = mail.fetch(e_id, "(RFC822)")
    if status != "OK":
        continue

    # Парсим письмо
    raw_email = msg_data[0][1]
    msg = email.message_from_bytes(raw_email)

    # Декодируем тему письма
    subject, encoding = decode_header(msg["Subject"])[0]
    if isinstance(subject, bytes):
        subject = subject.decode(encoding if encoding else "utf-8")
    
    # Очищаем тему от недопустимых символов для имени файла
    safe_subject = re.sub(r'[\\/*?:"<>|]', "_", subject)
    if not safe_subject:
        safe_subject = "no_subject"

    # Формируем имя файла
    filename = os.path.join(OUTPUT_DIR, f"{i:04d}_{safe_subject}.txt")

    # Извлекаем текст письма
    body = ""
    if msg.is_multipart():
        for part in msg.walk():
            content_type = part.get_content_type()
            content_disposition = str(part.get("Content-Disposition"))
            # Пропускаем вложения и берём только текст
            if content_type == "text/plain" and "attachment" not in content_disposition:
                try:
                    body = part.get_payload(decode=True).decode("utf-8", errors="ignore")
                except:
                    body = part.get_payload(decode=True).decode("latin-1", errors="ignore")
                break
    else:
        body = msg.get_payload(decode=True).decode("utf-8", errors="ignore")

    # Записываем в файл
    with open(filename, "w", encoding="utf-8") as f:
        f.write(f"Тема: {subject}\n")
        f.write(f"От: {msg.get('From')}\n")
        f.write(f"Дата: {msg.get('Date')}\n")
        f.write("-" * 40 + "\n\n")
        f.write(body)
    
    print(f"[{i}/{len(email_ids)}] Сохранено: {filename}")

# --- ЗАВЕРШЕНИЕ РАБОТЫ ---
mail.logout()
print("🏁 Готово! Все письма сохранены.")
```

💡 Пояснения к коду

· Подключение: Используется imaplib.IMAP4_SSL, что обеспечивает шифрование трафика. Если ваш сервер требует обычное соединение (без SSL), замените на imaplib.IMAP4 и порт 143.
· Поиск писем: Команда mail.search(None, "ALL") возвращает ID всех писем в папке. Если нужно выгрузить только непрочитанные, используйте "UNSEEN".
· Декодирование: Тема письма может быть закодирована, поэтому используется decode_header. Тело письма может быть в разных кодировках, скрипт пытается декодировать как utf-8, при ошибке — как latin-1.
· Сохранение: Каждое письмо сохраняется в отдельный файл. Имя файла формируется из порядкового номера и темы (недопустимые символы заменяются на _).

🏢 Особенности для разных корпоративных почт

· Яндекс 360: Сервер imap.yandex.ru, порт 993. В настройках почты нужно разрешить доступ по IMAP.
· Mail.ru для бизнеса: Сервер imap.mail.ru, порт 993.
· Microsoft Exchange / Office 365: Сервер outlook.office365.com, порт 993. Важно: Microsoft постепенно отказывается от базовой аутентификации по паролю в пользу OAuth 2.0. Если у вас современный тенант, обычный пароль может не сработать — потребуется регистрация приложения в Azure AD и получение токена доступа. В этом случае можно использовать библиотеку msal и exchangelib, но это отдельная более сложная тема.
· Корпоративный сервер без SSL: Если IT-отдел сообщил, что используется незащищённое соединение (порт 143), используйте imaplib.IMAP4 вместо IMAP4_SSL.
